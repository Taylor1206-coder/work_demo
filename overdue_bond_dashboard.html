특수채권 단계별 집계 쿼리 — d_debt_relief + d_debt_relief_status_history▸
-- 특수채권 유형별 + 최신 상태별 집계
-- 최신 status: registered_datetime MAX 기준 (= 현재 진행 단계)
-- [매각 건 제외 — 기본] WHERE 절 마지막 AND NOT EXISTS 조건 적용
-- [매각 건 포함 — 토글 ON] 해당 AND NOT EXISTS 조건 제거
WITH latest_status AS (
    SELECT
        d_debt_relief_uid,
        status,
        registered_datetime,
        ROW_NUMBER() OVER (
            PARTITION BY d_debt_relief_uid
            ORDER BY registered_datetime DESC
        ) AS rn
    FROM d_debt_relief_status_history
)
SELECT
    CASE dr.type
        WHEN 1 THEN '개인회생'
        WHEN 2 THEN '일반회생'
        WHEN 3 THEN '신용회복'
        WHEN 4 THEN '파산면책'
        ELSE '사망/기타'
    END                                          AS relief_type,
    ls.status                                    AS latest_status,
    CASE ls.status
        WHEN '110' THEN '접수/금지명령'
        WHEN '120' THEN '개시결정'
        WHEN '130' THEN '인가확정'
        ELSE         ls.status
    END                                          AS status_label,
    COUNT(DISTINCT dr.uid)                       AS case_cnt,
    COUNT(DISTINCT b.uid)                        AS bond_cnt,
    SUM(b.amount - COALESCE(paid.total_paid, 0)) AS outstanding_sum
FROM d_debt_relief dr
JOIN latest_status ls
    ON dr.uid = ls.d_debt_relief_uid AND ls.rn = 1
JOIN p_deal d        ON dr.deal_uid = d.deal_uid
JOIN tb_lo_bond b    ON d.deal_uid = b.deal_uid
LEFT JOIN (
    SELECT bond_uid, SUM(principal) AS total_paid
    FROM b_repayment_paid GROUP BY bond_uid
) paid ON b.uid = paid.bond_uid
WHERE
    dr.deleted_datetime IS NULL
    AND b.overdue_start_date IS NOT NULL
    AND NOT EXISTS (
        SELECT 1 FROM b_repayment_sale s
        WHERE s.bond_uid = b.uid AND s.is_whole_sale = 1)
GROUP BY dr.type, ls.status
ORDER BY dr.type, ls.status;

SELECT
    d.deal_uid,
    b.uid                                    AS bond_uid,
    b.amount                                 AS loan_amount,
    b.amount - COALESCE(paid.total_paid, 0)   AS outstanding_balance,
    b.overdue_start_date,
    DATEDIFF(CURDATE(), b.overdue_start_date)   AS overdue_days,
    drh.status                               AS relief_status,
    CASE
        WHEN drh.status = '130'
            THEN ROUND((b.amount - COALESCE(paid.total_paid,0)) * 0.80, 0)
        WHEN drh.status IN ('120', '110')
            THEN ROUND((b.amount - COALESCE(paid.total_paid,0)) * 0.34, 0)
        WHEN dr.uid IS NULL
         AND DATEDIFF(CURDATE(), b.overdue_start_date) >= 30
            THEN ROUND((b.amount - COALESCE(paid.total_paid,0)) * 0.25, 0)
        ELSE NULL
    END                                      AS expected_sale_price
FROM tb_lo_bond b
JOIN p_deal d           ON b.deal_uid = d.deal_uid
JOIN tb_lo_apply la     ON d.loan_apply_uid = la.uid
JOIN l_credit_apply ca  ON la.uid = ca.apply_uid
LEFT JOIN ( SELECT bond_uid, SUM(principal) AS total_paid
            FROM b_repayment_paid GROUP BY bond_uid ) paid
    ON b.uid = paid.bond_uid
LEFT JOIN d_debt_relief dr
    ON d.deal_uid = dr.deal_uid
   AND dr.type = 1 AND dr.deleted_datetime IS NULL
LEFT JOIN d_debt_relief_status_history drh
    ON dr.uid = drh.d_debt_relief_uid
   AND drh.registered_datetime = (
       SELECT MAX(h.registered_datetime)
       FROM d_debt_relief_status_history h
       WHERE h.d_debt_relief_uid = dr.uid )
WHERE b.overdue_start_date IS NOT NULL
  AND NOT EXISTS (
      SELECT 1 FROM b_repayment_sale brs
      WHERE brs.bond_uid = b.uid AND brs.is_whole_sale = 1 )
ORDER BY overdue_days DESC;

① 연체 전이율 (Delinquency Migration Rate)
M1(~30일) → M2(31~60일) → M3(61~90일) → M4(90일+) · 1개월 단위 미상환 잔존 비율
기준: analysis.bond.daily_snapshot (월말 스냅샷 비교)
📐 지표 정의
연체 전이율은 동일 채권이 시간 흐름에 따라 연체 구간을 악화하는 비율입니다.
전월말 스냅샷에서 M1 구간(연체 1~30일)에 있던 채권이 당월말 스냅샷에서 M2(31~60일)로 이동했다면 → M1→M2 전이.
즉 M1 → M2 → M3는 시간 경과에 따른 연체 심화를 나타내며, M1이 가장 초기(최근 연체), M4+가 가장 심각(장기연체)입니다.
모수: 전월말 스냅샷 기준 구간 내 채권
분자: 당월말에 상위 구간으로 이동한 채권 수
데이터 소스: analysis.bond.daily_snapshot (월말 스냅샷)

-- 전이율: 전월말 M버킷 → 당월말 M버킷 비교 (daily_snapshot 활용)
WITH
prev_month AS (
    -- 전월말 스냅샷에서 각 채권의 연체 구간 확정
    SELECT
        bond_uid,
        outstanding_balance,
        CASE
            WHEN overdue_days BETWEEN 1  AND 30 THEN 'M1'
            WHEN overdue_days BETWEEN 31 AND 60 THEN 'M2'
            WHEN overdue_days BETWEEN 61 AND 90 THEN 'M3'
            WHEN overdue_days > 90           THEN 'M4+'
            ELSE '정상'
        END AS bucket
    FROM analysis.bond.daily_snapshot
    WHERE snapshot_date = LAST_DAY(DATE_SUB(CURDATE(), INTERVAL 1 MONTH))
      AND is_whole_sale = 0
),
curr_month AS (
    -- 당월말(또는 최신) 스냅샷
    SELECT
        bond_uid,
        outstanding_balance,
        CASE
            WHEN overdue_days BETWEEN 1  AND 30 THEN 'M1'
            WHEN overdue_days BETWEEN 31 AND 60 THEN 'M2'
            WHEN overdue_days BETWEEN 61 AND 90 THEN 'M3'
            WHEN overdue_days > 90           THEN 'M4+'
            ELSE '정상'
        END AS bucket
    FROM analysis.bond.daily_snapshot
    WHERE snapshot_date = LAST_DAY(CURDATE())  -- 또는 MAX(snapshot_date)
      AND is_whole_sale = 0
)
SELECT
    p.bucket                              AS prev_bucket,
    c.bucket                              AS curr_bucket,
    COUNT(*)                              AS cnt,
    SUM(c.outstanding_balance)            AS outstanding_sum,
    ROUND(COUNT(*) * 100.0
        / SUM(COUNT(*)) OVER (PARTITION BY p.bucket), 1)
                                          AS transition_rate_pct
FROM prev_month p
JOIN curr_month c ON p.bond_uid = c.bond_uid
WHERE p.bucket IN ('M1', 'M2', 'M3')  -- 전월 연체 구간이 모수
GROUP BY p.bucket, c.bucket
ORDER BY p.bucket, c.bucket;
② 연체 해소율 (Delinquency Resolution Rate)
연체 발생 코호트 기준 · D+30/60/90 시점 잔존율 추적 · overdue_start_date 기반 생존 곡선
기준: tb_lo_bond.overdue_start_date + b_repayment_paid
📐 지표 정의
연체 해소율은 특정 시점에 연체가 발생한 코호트(동일 월 연체 시작 채권)가 이후 각 시점까지 상환 완료된 비율입니다.
예: 2026-03에 연체가 시작된 채권 N건 중 → D+30일 시점에 몇 건이 해소됐는가?
전이율과 반대 방향: D+30 → D+60 → D+90으로 갈수록 누적 해소 비율 증가, 잔존 비율 감소.
코호트: 같은 달 overdue_start_date 채권 묶음
해소 기준: b_repayment_paid 기록 D+N일 이전 존재
시점: D+30 / D+60 / D+90 / D+180
💡 전이율(①)이 얼마나 빠르게 악화되는지를 본다면, 해소율은 얼마나 빠르게 회복되는지를 봅니다.

연체 해소율 쿼리 — 코호트별 생존 추적 (평균 금리/등급 포함)▸
-- 코호트 = overdue_start_date의 월 기준 · 평균 금리/HF 등급 조인 추가
WITH
cohort AS (
    SELECT
        b.uid                                        AS bond_uid,
        DATE_FORMAT(b.overdue_start_date, '%Y-%m')  AS cohort_month,
        b.overdue_start_date,
        ca.rate                                      AS loan_rate,
        lc.hf_grade_final                            AS hf_grade
    FROM tb_lo_bond b
    JOIN p_deal d           ON b.deal_uid = d.deal_uid
    JOIN tb_lo_apply la     ON d.loan_apply_uid = la.uid
    JOIN l_credit_apply ca  ON la.uid = ca.apply_uid
    LEFT JOIN l_limit_calculation lc ON la.uid = lc.credit_apply_uid
    WHERE
        b.overdue_start_date IS NOT NULL
        AND la.loan_product_type = 1
        AND NOT EXISTS (
            SELECT 1 FROM b_repayment_sale s
            WHERE s.bond_uid = b.uid AND s.is_whole_sale = 1)
),
resolution AS (
    SELECT bond_uid, MIN(paid_date) AS first_paid_date
    FROM b_repayment_paid GROUP BY bond_uid
)
SELECT
    c.cohort_month,
    COUNT(*)                                                AS total_cnt,
    ROUND(AVG(c.loan_rate) * 100, 2)                    AS avg_rate_pct,
    ROUND(AVG(c.hf_grade), 1)                             AS avg_hf_grade,
    SUM(CASE WHEN r.first_paid_date > c.overdue_start_date + INTERVAL 30 DAY OR r.first_paid_date IS NULL THEN 1 ELSE 0 END) AS remain_d30,
    ROUND(SUM(CASE WHEN r.first_paid_date <= c.overdue_start_date + INTERVAL 30 DAY THEN 1 ELSE 0 END) * 100.0 / COUNT(*), 1) AS resolution_rate_d30,
    SUM(CASE WHEN r.first_paid_date > c.overdue_start_date + INTERVAL 60 DAY OR r.first_paid_date IS NULL THEN 1 ELSE 0 END) AS remain_d60,
    ROUND(SUM(CASE WHEN r.first_paid_date <= c.overdue_start_date + INTERVAL 60 DAY THEN 1 ELSE 0 END) * 100.0 / COUNT(*), 1) AS resolution_rate_d60,
    SUM(CASE WHEN r.first_paid_date > c.overdue_start_date + INTERVAL 90 DAY OR r.first_paid_date IS NULL THEN 1 ELSE 0 END) AS remain_d90,
    ROUND(SUM(CASE WHEN r.first_paid_date <= c.overdue_start_date + INTERVAL 90 DAY THEN 1 ELSE 0 END) * 100.0 / COUNT(*), 1) AS resolution_rate_d90,
    SUM(CASE WHEN r.first_paid_date > c.overdue_start_date + INTERVAL 180 DAY OR r.first_paid_date IS NULL THEN 1 ELSE 0 END) AS remain_d180,
    ROUND(SUM(CASE WHEN r.first_paid_date <= c.overdue_start_date + INTERVAL 180 DAY THEN 1 ELSE 0 END) * 100.0 / COUNT(*), 1) AS resolution_rate_d180
FROM cohort c
LEFT JOIN resolution r ON c.bond_uid = r.bond_uid
GROUP BY c.cohort_month
ORDER BY c.cohort_month DESC
LIMIT 12;

30일 상환완료율 쿼리▸
SET @base_date = DATE_SUB(CURDATE(), INTERVAL 1 DAY);  -- 전일
SET @overdue_date = DATE_SUB(@base_date, INTERVAL 30 DAY); -- 30일 전 연체 시작일

SELECT
    COUNT(*) AS total_overdue_30d,
    SUM(CASE WHEN repaid.bond_uid IS NOT NULL THEN 1 ELSE 0 END) AS repaid_cnt,
    ROUND(
        SUM(CASE WHEN repaid.bond_uid IS NOT NULL THEN 1 ELSE 0 END) * 100.0
        / COUNT(*), 1
    ) AS repayment_rate_pct
FROM tb_lo_bond b
LEFT JOIN (
    -- 30일 이내 기간 중 상환 기록 존재 여부
    SELECT DISTINCT bond_uid
    FROM b_repayment_paid
    WHERE paid_date BETWEEN @overdue_date AND @base_date
) repaid ON b.uid = repaid.bond_uid
WHERE
    b.overdue_start_date <= @overdue_date   -- 30일 이상 연체 지속 채권 (기간 지속 포함)
    AND NOT EXISTS (
        SELECT 1 FROM b_repayment_sale s
        WHERE s.bond_uid = b.uid AND s.is_whole_sale = 1);

단기 연체 반복 쿼리 (회차별 지연일수 기반)▸
-- b_repayment_charged: actual_repayment_date - adjusted_repayment_date = 지연일수
SELECT
    rc.bond_uid,
    COUNT(*) AS late_seq_count,       -- 지연 발생 회차 수
    AVG(delay_days)  AS avg_delay_days,
    SUM(rp.repaid_amt) AS total_repaid,
    AVG(rp.repaid_amt) AS avg_repaid_per_seq,
    b.amount          AS loan_amount,
    ROUND(AVG(rp.repaid_amt) / b.amount * 100, 1) AS avg_repaid_pct
FROM (
    SELECT
        bond_uid,
        deal_repayment_seq,
        DATEDIFF(actual_repayment_date, adjusted_repayment_date) AS delay_days
    FROM b_repayment_charged
    WHERE
        actual_repayment_date IS NOT NULL
        AND actual_repayment_date > adjusted_repayment_date
        AND DATEDIFF(actual_repayment_date, adjusted_repayment_date) < 30 -- 단기만
) rc
JOIN tb_lo_bond b ON rc.bond_uid = b.uid
LEFT JOIN (
    SELECT bond_uid, deal_repayment_seq,
           SUM(principal + interest + late_fee) AS repaid_amt
    FROM b_repayment_paid
    GROUP BY bond_uid, deal_repayment_seq
) rp ON rc.bond_uid = rp.bond_uid AND rc.deal_repayment_seq = rp.deal_repayment_seq
GROUP BY rc.bond_uid, b.amount
HAVING late_seq_count >= 2       -- 2회 이상 반복
ORDER BY late_seq_count DESC
LIMIT 50;
⑤ 개인회생 채권 대출실행 월 분포
d_debt_relief(type=1) · 과거 12개월 당월 제외 (2025-06~2026-05) · 기표월별 건수 / 평균 금리 / 등급
기표월 (tb_lo_bond.start_date) 건수 평균 금리 (%) 평균 HF 등급 평균 대출금액 개회생 비율
Join 경로: d_debt_relief → p_deal → tb_lo_bond.start_date (기표일) / tb_lo_apply → l_credit_apply.rate (금리) / l_limit_calculation.hf_grade_final (최종 등급)
개인회생 기표월 분포 쿼리▸
SELECT
    DATE_FORMAT(b.start_date, '%Y-%m')         AS origination_month,
    COUNT(*)                                   AS cnt,
    ROUND(AVG(ca.rate) * 100, 2)            AS avg_rate_pct,
    ROUND(AVG(lc.hf_grade_final), 1)         AS avg_hf_grade,
    ROUND(AVG(b.amount), 0)                 AS avg_loan_amount
FROM d_debt_relief dr
JOIN p_deal d          ON dr.deal_uid = d.deal_uid
JOIN tb_lo_bond b      ON b.deal_uid = d.deal_uid
JOIN tb_lo_apply la    ON d.loan_apply_uid = la.uid
LEFT JOIN l_credit_apply ca
    ON la.uid = ca.apply_uid
LEFT JOIN l_limit_calculation lc
    ON ca.apply_uid = lc.credit_apply_uid
WHERE
    dr.type = 1
    AND dr.deleted_datetime IS NULL
    AND la.loan_product_type = 1   -- 개인신용대출
GROUP BY origination_month
ORDER BY origination_month;

당월 연체채권 기표월 분포 쿼리▸
SELECT
    DATE_FORMAT(b.start_date, '%Y-%m') AS origination_month,
    COUNT(*)                          AS overdue_cnt,
    SUM(b.amount - COALESCE(paid.total_paid, 0))
                                      AS outstanding_sum,
FROM tb_lo_bond b
JOIN p_deal d        ON b.deal_uid = d.deal_uid
JOIN tb_lo_apply la  ON d.loan_apply_uid = la.uid
LEFT JOIN (
    SELECT bond_uid, SUM(principal) AS total_paid
    FROM b_repayment_paid GROUP BY bond_uid
) paid ON b.uid = paid.bond_uid
WHERE
    b.overdue_start_date IS NOT NULL
    AND la.loan_product_type = 1
    AND NOT EXISTS (
        SELECT 1 FROM b_repayment_sale s
        WHERE s.bond_uid = b.uid AND s.is_whole_sale = 1)
GROUP BY origination_month
ORDER BY origination_month;
